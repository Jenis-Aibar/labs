CREATE OR REPLACE FUNCTION partSearch(parts TEXT)
RETURNS TABLE(id BIGINT, firstname VARCHAR, lastname VARCHAR, contact VARCHAR) AS $$
BEGIN
    RETURN QUERY
    SELECT id, firstname, lastname, contact
    FROM mainData
    WHERE firstname ILIKE '%' || parts || '%' 
       OR lastname ILIKE '%' || parts || '%'
       OR contact LIKE '%' || parts || '%';
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE PROCEDURE addChangeUsers(qfirstname VARCHAR, qlastname VARCHAR, qcontact VARCHAR)
AS $$
BEGIN
    IF EXISTS (SELECT 1 FROM mainData WHERE firstname = qfirstname AND lastname = qlastname) THEN
        UPDATE mainData SET contact = qcontact WHERE firstname = qfirstname AND lastname = qlastname;
    ELSE
        INSERT INTO mainData(firstname, lastname, contact) VALUES (qfirstname, qlastname, qcontact);
    END IF;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE PROCEDURE addMultipleUsers(qusers TEXT[])
AS $$
DECLARE
    quser RECORD;
    invalid_users TEXT[] := '{}';
BEGIN
    FOREACH quser IN ARRAY qusers
    LOOP
        IF quser.contact ~ '^87[0-9]{9}$' THEN
            INSERT INTO mainData(firstname, lastname, contact) VALUES (quser.firstname, quser.lastname, quser.contact);
        ELSE
            invalid_users := array_append(invalid_users, quser.firstname || ' ' || quser.lastname || ' - ' || quser.contact);
        END IF;
    END LOOP;

    IF array_length(invalid_users, 1) > 0 THEN
        RAISE NOTICE 'Invalid users: %', invalid_users;
    END IF;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION getPagination(p_limit INT, p_offset INT)
RETURNS TABLE(id BIGINT, firstname VARCHAR, lastname VARCHAR, contact VARCHAR) AS $$
BEGIN
    RETURN QUERY
    SELECT id, firstname, lastname, contact
    FROM mainData
    LIMIT p_limit OFFSET p_offset;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE PROCEDURE deleteUser(p_firstname VARCHAR, p_lastname VARCHAR, p_contact VARCHAR)
AS $$
BEGIN
    IF p_firstname IS NOT NULL AND p_lastname IS NOT NULL THEN
        DELETE FROM mainData WHERE firstname = p_firstname AND lastname = p_lastname;
    END IF;

    IF p_contact IS NOT NULL THEN
        DELETE FROM mainData WHERE contact = p_contact;
    END IF;
END;
$$ LANGUAGE plpgsql;
